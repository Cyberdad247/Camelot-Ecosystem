    <script>
        const { jsPDF } = window.jspdf;

        const questions = [
            { id: 'speaker', q: "High-ticket hardware leads leaking because response latency exceeds 5 minutes?", deliverable: "AI Sales-Bot (Speaker Technical Support Tier)", price: 2000, day: 30 },
            { id: 'inventory', q: "Clothing drop fulfillment managed manually with zero autonomous flow?", deliverable: "Autonomous Apparel Inventory & Fulfillment matrix", price: 2500, day: 60 },
            { id: 'search', q: "Brand invisible to AI search engines for premium speakers?", deliverable: "360 GEO (Generative Engine Optimization) Campaign", price: 3500, day: 60 },
            { id: 'autonomy', q: "Business cannot run for 30 days without constant human babysitting?", deliverable: "Camelot-OS Infrastructure Integration", price: 5000, day: 90 }
        ];

        let state = { answers: {}, price: 0, signedAt: null, location: "Cleveland, OH", sid: 'SID-' + Math.random().toString(36).substring(2, 9).toUpperCase() };
        document.getElementById('session-id').innerText = state.sid;

        // Clock Update
        setInterval(() => {
            document.getElementById('live-clock').innerText = new Date().toLocaleTimeString('en-US', { hour12: false });
        }, 1000);

        // Audit Gauge Animation
        setTimeout(() => {
            document.getElementById('g-seo').style.width = '42%';
            document.getElementById('g-geo').style.width = '18%';
            document.getElementById('g-leak').style.width = '15%';
            document.getElementById('g-vel').style.width = '65%';
        }, 500);

        // Diagnostic Grid Render
        const grid = document.getElementById('diag-grid');
        questions.forEach(q => {
            const card = document.createElement('div');
            card.className = "p-8 rounded-[40px] bento-card space-y-8 cursor-default";
            card.innerHTML = `<p class="text-white text-xl font-black h-16 leading-tight italic">${q.q}</p>
                <div class="flex gap-4"><button class="ans-btn flex-1 py-4 bg-white/5 rounded-2xl font-black text-[10px] uppercase tracking-widest border border-white/5" onclick="handleAns('${q.id}', true, this)">YES</button>
                <button class="ans-btn flex-1 py-4 bg-white/5 rounded-2xl font-black text-[10px] uppercase tracking-widest border border-white/5" onclick="handleAns('${q.id}', false, this)">NO</button></div>`;
            grid.appendChild(card);
        });

        function handleAns(id, val, btn) {
            btn.parentElement.querySelectorAll('button').forEach(b => b.classList.remove('bg-kba-purple', 'bg-gray-800', 'text-white'));
            btn.classList.add(val ? 'bg-kba-purple' : 'bg-gray-800');
            state.answers[id] = val;
            if (Object.keys(state.answers).length >= 1) document.getElementById('initialize-vault').classList.remove('hidden');
        }

        function changePhase(phaseId) {
            if (phaseId === 'vault' && document.getElementById('nav-vault').classList.contains('opacity-40')) {
                alert("ACCESS DENIED: Complete Discovery Inquisition to unlock Vault."); return;
            }
            document.querySelectorAll('.phase-content').forEach(c => c.classList.remove('active'));
            document.getElementById(`phase-${phaseId}`).classList.add('active');
            document.querySelectorAll('.phase-nav-btn').forEach(b => b.classList.remove('active'));
            document.getElementById(`nav-${phaseId}`).classList.add('active');
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        // Vault Forge
        document.getElementById('initialize-vault').addEventListener('click', () => {
            document.getElementById('nav-vault').classList.remove('opacity-40', 'cursor-not-allowed');
            changePhase('vault');
            const list = document.getElementById('sow-list');
            const roadmap = document.getElementById('roadmap-list');
            list.innerHTML = ''; roadmap.innerHTML = '';
            state.price = 0;

            questions.forEach(q => {
                if (state.answers[q.id]) {
                    state.price += q.price;
                    const item = document.createElement('div');
                    item.className = "p-5 bg-white/5 border-l-4 border-kba-gold rounded-r-xl text-[9px] font-black uppercase tracking-widest text-gray-300 shadow-xl";
                    item.innerText = q.deliverable;
                    list.appendChild(item);

                    const roadItem = document.createElement('p');
                    roadItem.innerHTML = `[Day ${q.day}] > ${q.deliverable}`;
                    roadmap.appendChild(roadItem);
                }
            });
            document.getElementById('price-total').innerText = `$${state.price.toLocaleString()}`;
            document.getElementById('price-down').innerText = `$${(state.price * 0.25).toLocaleString()}`;
        });

        // Tab System
        const docsViewed = new Set();
        document.querySelectorAll('.tab-btn').forEach(btn => {
            btn.addEventListener('click', () => {
                document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
                document.querySelectorAll('.doc-tab-content').forEach(c => c.classList.remove('active'));
                btn.classList.add('active');
                document.getElementById(btn.getAttribute('data-target')).classList.add('active');
                docsViewed.add(btn.getAttribute('data-target'));
                if (docsViewed.size >= 5) document.getElementById('handshake-btn').classList.remove('opacity-40', 'cursor-not-allowed');
            });
        });

        // Signature
        const canvas = document.getElementById('sig-canvas');
        const ctx = canvas.getContext('2d');
        let drawing = false;
        canvas.addEventListener('mousedown', () => drawing = true);
        canvas.addEventListener('mouseup', () => { drawing = false; ctx.beginPath(); });
        canvas.addEventListener('mousemove', (e) => {
            if (!drawing) return;
            const rect = canvas.getBoundingClientRect();
            ctx.lineWidth = 4; ctx.strokeStyle = '#000'; ctx.lineCap = 'round';
            ctx.lineTo(e.clientX - rect.left, e.clientY - rect.top); ctx.stroke();
        });
        function clearSig() { ctx.clearRect(0, 0, canvas.width, canvas.height); }

        // Finalize
        document.getElementById('finalize-all').addEventListener('click', () => {
            if (canvas.toDataURL() === document.createElement('canvas').toDataURL()) { alert("SIGNATURE REQUIRED."); return; }
            state.signedAt = new Date().toLocaleString('en-US', { timeZone: 'America/New_York' });
            document.querySelectorAll('.doc-tab-content').forEach(c => c.classList.remove('active'));
            document.getElementById('artifact-box').classList.remove('hidden');
        });

        // PDF certified
        function downloadCertification() {
            const doc = new jsPDF();
            doc.setFont("courier", "bold"); doc.setFontSize(22);
            doc.text("HANDSHAKE CERTIFICATE OF AUTHENTICITY", 20, 30);
            doc.setFontSize(10); doc.setFont("courier", "normal");
            doc.text(`Digital Session ID: ${state.sid}`, 20, 50);
            doc.text(`Location Verified: Cleveland, OH`, 20, 56);
            doc.text(`Handshake Certified (EST): ${state.signedAt}`, 20, 62);
            doc.line(20, 75, 190, 75);
            doc.text("GOVERNANCE ACTIVE: 48-Hour SLA | Non-Refundable Deposit", 20, 85);
            const imgData = canvas.toDataURL('image/png');
            doc.addImage(imgData, 'PNG', 20, 110, 100, 40);
            doc.text("AUTHORIZED CEO SIGNATURE (ELECTRONICALLY SEALED)", 20, 160);
            doc.save(`Certification_${state.sid}.pdf`);
        }

        function downloadFullPackage() {
            const doc = new jsPDF();
            doc.setFontSize(20); doc.text("MASTER BLUEPRINT ARCHIVE", 20, 30);
            doc.setFontSize(10); let y = 50;
            doc.text("PART A: REFORGED DELIVERABLES", 20, y); y += 12;
            questions.forEach(q => { if(state.answers[q.id]) { doc.text(`[✓] ${q.deliverable}`, 25, y); y += 8; } });
            y += 10; doc.text(`Authorized Total Stake: $${state.price.toLocaleString()}`, 20, y);
            y += 20; doc.text("PART B: MASTER GOVERNANCE", 20, y); y += 10;
            doc.text("- 48-hour mockup sign-off window active.", 20, y);
            doc.save(`Full_Package_${state.sid}.pdf`);
        }
